<script>
const ler = chave => {
  try { return JSON.parse(localStorage.getItem(chave)) || []; }
  catch { return []; }
};
const dados = {
  financeiro: ler("gf_financeiro"),
  veiculos: ler("gf_veiculos"),
  viagens: ler("gf_viagens")
};
const salvar = () => {
  Object.entries(dados).forEach(([k,v]) =>
    localStorage.setItem("gf_"+k,v ? JSON.stringify(v) : "[]")
  );
  renderizar();
};
const el = id => document.getElementById(id);
const dinheiro = n => Number(n || 0).toLocaleString("pt-BR",{style:"currency",currency:"BRL"});
const seguro = v => String(v ?? "").replace(/[&<>"']/g,c=>({
  "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
}[c]));
const idNovo = () => crypto.randomUUID ? crypto.randomUUID() : String(Date.now()+Math.random());

document.querySelectorAll("nav button").forEach(btn=>{
  btn.addEventListener("click",()=>{
    document.querySelectorAll("nav button").forEach(b=>b.classList.toggle("active",b===btn));
    document.querySelectorAll("main section").forEach(s=>s.classList.toggle("hidden",s.id!==btn.dataset.tab));
  });
});

el("formFinanceiro").addEventListener("submit",e=>{
  e.preventDefault();
  dados.financeiro.push({
    id:idNovo(),tipo:el("tipo").value,descricao:el("descricao").value.trim(),
    pessoa:el("pessoa").value.trim(),valor:Math.round(Number(el("valor").value)*100)/100,
    vencimento:el("vencimento").value,situacao:el("situacao").value
  });
  e.target.reset(); salvar();
});
el("formVeiculo").addEventListener("submit",e=>{
  e.preventDefault();
  const placa=el("placa").value.trim().toUpperCase();
  if(placa && dados.veiculos.some(v=>v.placa===placa)){
    alert("Já existe um veículo com essa placa.");return;
  }
  dados.veiculos.push({id:idNovo(),nome:el("nomeVeiculo").value.trim(),tipo:el("tipoVeiculo").value,placa});
  e.target.reset(); salvar();
});
el("formViagem").addEventListener("submit",e=>{
  e.preventDefault();
  dados.viagens.push({
    id:idNovo(),data:el("dataViagem").value,veiculoId:el("veiculoViagem").value,
    origem:el("origem").value.trim(),destino:el("destino").value.trim(),
    produto:el("produto").value.trim(),toneladas:Number(el("toneladas").value||0),
    frete:Math.round(Number(el("frete").value||0)*100)/100
  });
  e.target.reset(); salvar();
});
function remover(grupo,id){
  if(!confirm("Deseja excluir este registro?"))return;
  dados[grupo]=dados[grupo].filter(x=>x.id!==id); salvar();
}
function acao(grupo,id){
  return `<button class="delete" data-remover="${grupo}" data-id="${id}">Excluir</button>`;
}
function renderizar(){
  const soma=tipo=>dados.financeiro.filter(x=>x.tipo===tipo).reduce((s,x)=>s+Number(x.valor),0);
  el("receitas").textContent=dinheiro(soma("receita"));
  el("despesas").textContent=dinheiro(soma("despesa"));
  const saldo=soma("receita")-soma("despesa");
  el("saldo").textContent=dinheiro(saldo);
  el("saldo").className=saldo<0?"negative":"positive";
  el("totalVeiculos").textContent=dados.veiculos.length;
  el("ultimos").innerHTML=dados.financeiro.length
    ? [...dados.financeiro].reverse().slice(0,5).map(x=>
      `<p>${seguro(x.descricao)} — ${dinheiro(x.valor)} (${seguro(x.situacao)})</p>`).join("")
    : "Nenhum lançamento registrado.";
  el("listaFinanceira").innerHTML=dados.financeiro.map(x=>`<tr>
    <td>${seguro(x.tipo)}</td><td>${seguro(x.descricao)}<br><small>${seguro(x.pessoa)}</small></td>
    <td>${dinheiro(x.valor)}</td><td>${seguro(x.vencimento||"—")}</td>
    <td>${seguro(x.situacao)}</td><td>${acao("financeiro",x.id)}</td></tr>`).join("");
  el("listaVeiculos").innerHTML=dados.veiculos.map(x=>`<tr>
    <td>${seguro(x.nome)}</td><td>${seguro(x.tipo)}</td><td>${seguro(x.placa||"—")}</td>
    <td>${acao("veiculos",x.id)}</td></tr>`).join("");
  el("veiculoViagem").innerHTML=dados.veiculos.map(x=>
    `<option value="${seguro(x.id)}">${seguro(x.nome)}${x.placa?" — "+seguro(x.placa):""}</option>`).join("");
  el("listaViagens").innerHTML=dados.viagens.map(x=>{
    const v=dados.veiculos.find(v=>v.id===x.veiculoId);
    return `<tr><td>${seguro(x.data)}</td><td>${seguro(v?.nome||"Veículo removido")}</td>
    <td>${seguro(x.origem)} → ${seguro(x.destino)}</td><td>${seguro(x.produto)}</td>
    <td>${x.toneladas}</td><td>${dinheiro(x.frete)}</td><td>${acao("viagens",x.id)}</td></tr>`;
  }).join("");
}
document.addEventListener("click",e=>{
  const b=e.target.closest("[data-remover]");
  if(b)remover(b.dataset.remover,b.dataset.id);
});
renderizar();
</script>
